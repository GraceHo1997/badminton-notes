-- 羽球筆記 v7 -- 請在 Supabase SQL Editor 執行一次。 -- v7 沿用 v6 的 match_order / match_invites，不需重建。 -- 好友查看紀錄的單一 RPC： -- 1. 必須是好友 -- 2. target 的 privacy.records 必須不是 private -- 3. 回傳總戰績、最近3筆、最佳隊友、最大敵手 create or replace function public.get_friend_record_overview(target uuid) returns jsonb language plpgsql security definer set search_path = public as $$ declare caller uuid := auth.uid(); p jsonb; is_friend boolean := false; total_count integer := 0; win_count integer := 0; loss_count integer := 0; rate numeric := 0; best_tm jsonb := null; top_rival jsonb := null; recent_rows jsonb := '[]'::jsonb; begin if caller is null then return jsonb_build_object('visible', false); end if; select privacy into p from public.profiles where id = target; select exists( select 1 from public.friend_requests fr where fr.status = 'accepted' and ( (fr.from_user = caller and fr.to_user = target) or (fr.from_user = target and fr.to_user = caller) ) ) into is_friend; if not is_friend then return jsonb_build_object('visible', false); end if; if coalesce(p->>'records','friends') = 'private' then return jsonb_build_object('visible', false); end if; select count(*), count(*) filter (where is_win = true), count(*) filter (where is_win = false) into total_count, win_count, loss_count from public.matches where user_id = target; if total_count > 0 then rate := round((win_count::numeric / total_count::numeric) * 100, 1); end if; -- 最佳隊友：雙打 teammate_name，先比勝率，再比場次。 select jsonb_build_object( 'name', teammate_name, 'games', count(*), 'wins', count(*) filter (where is_win = true), 'win_rate', round((count(*) filter (where is_win = true))::numeric / count(*)::numeric * 100, 1) ) into best_tm from public.matches where user_id = target and type = '雙打' and nullif(trim(teammate_name),'') is not null group by teammate_name order by ((count(*) filter (where is_win = true))::numeric / count(*)::numeric) desc, count(*) desc, teammate_name limit 1; -- 最大敵手：作為 opponent / opponent2 出現次數最多的人。 with rivals as ( select opponent as name from public.matches where user_id = target and nullif(trim(opponent),'') is not null union all select opponent2_name as name from public.matches where user_id = target and type='雙打' and nullif(trim(opponent2_name),'') is not null ) select jsonb_build_object('name', name, 'games', count(*)) into top_rival from rivals group by name order by count(*) desc, name limit 1; select coalesce(jsonb_agg(x.obj order by x.match_date desc, x.match_order asc, x.game_no asc),'[]'::jsonb) into recent_rows from ( select m.match_date, coalesce(m.match_order, m.id::integer) as match_order, m.game_no, jsonb_build_object( 'date', m.match_date, 'type', case when m.type='雙打' and nullif(m.doubles_type,'') is not null then m.doubles_type else m.type end, 'win', m.is_win ) as obj from public.matches m where m.user_id = target order by m.match_date desc, coalesce(m.match_order, m.id::integer) asc, m.game_no asc limit 3 ) x; return jsonb_build_object( 'visible', true, 'stats', jsonb_build_object( 'total', total_count, 'wins', win_count, 'losses', loss_count, 'win_rate', rate ), 'best_teammate', best_tm, 'top_rival', top_rival, 'recent', recent_rows ); end; $$; grant execute on function public.get_friend_record_overview(uuid) to authenticated;
